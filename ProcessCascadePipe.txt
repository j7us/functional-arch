public static BoardState ProcessCascade(BoardState currentState)
        {
            var matches = FindMatches(currentState.Board);
            if (matches.Count == 0) return currentState;

            return currentState.Pipe(x -> RemoveMatches(x, matches))
            .Pipe(FillEmptySpaces)
            .Pipe(ProcessCascade)
        }

        public static TOut Pipe<TIn, TOut>(this TIn input, Func<TIn, TOut> func)
        {
                return func(input);
        }
